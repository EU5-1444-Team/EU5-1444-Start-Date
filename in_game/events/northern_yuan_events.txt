namespace = northern_yuan

# Yuan demands submission of all clans
northern_yuan.100 = {
    type = country_event
    title = northern_yuan.100.title
    desc = northern_yuan.100.desc

    illustration_tags = {
        10 = regular
        10 = interior
    }

    immediate = {
        event_illustration_estate_effect = { foreground = estate_type:nobles_estate background = estate_type:nobles_estate }        
    }

    option = {
        name = northern_yuan.100.a
        add_prestige = prestige_ultimate_bonus
        custom_tooltip = {
            text = northern_yuan.100.ctt
            international_organization:northern_yuan = {
                every_international_organization_member = {
                    limit = {
                        NOT = { root = this }
                        OR = {
                            is_subject = no
                            is_subject_type = tributary
                        }
                    }
                    trigger_event_silently = { id = northern_yuan.101 }
                }
            }
        }
    }
}

#... and for member states
northern_yuan.101 = {
    type = country_event
    title = northern_yuan.101.title
    desc = northern_yuan.101.desc
    illustration_tags = {
        10 = angry
        10 = interior
    }

    immediate = {
        event_illustration_estate_effect = { foreground = estate_type:nobles_estate background = estate_type:nobles_estate }
        
        international_organization:northern_yuan = {
            every_international_organization_member = {
                limit = {
                    has_special_status_in_international_organization = {
                        type = special_status:protector_general
                        international_organization = international_organization:northern_yuan
                    }
                }
                save_scope_as = northern_yuan_boss_scope
            }
        }
    }

    option = {
        name = northern_yuan.101.a
        ai_chance = {
            factor = 10
            modifier = {
                factor = 0
                opinion = {
                    target = scope:northern_yuan_boss_scope
                    value < 100
                }
            }
            # If I am Oirat, I refuse if the boss is NOT Oirat
            modifier = {
                factor = 0
                culture = culture:oirat_culture
                scope:northern_yuan_boss_scope = {
                    NOT = { culture = culture:oirat_culture }
                }
            }
            modifier = {
                factor = 0
                OR = {
                    is_subject_of = c:CHI
                    any_overlord_or_above = { tag = c:CHI } # Catch-all for different subject types
                }
            }
        }
        
        make_subject_of = {
            target = scope:northern_yuan_boss_scope
            type = subject_type:vassal
        }
    }

    option = {
        name = northern_yuan.101.b
        ai_chance = {
            factor = 10
            modifier = {
                factor = 0
                opinion = {
                    target = scope:northern_yuan_boss_scope
                    value >= 100
                }
            }
            modifier = {
                factor = 100
                culture = culture:oirat_culture
                scope:northern_yuan_boss_scope = {
                    NOT = { culture = culture:oirat_culture }
                }
            }
            modifier = {
                factor = 100
                OR = {
                    is_subject_of = c:CHI
                    any_overlord_or_above = { tag = c:CHI }
                }
            }
        }
        scope:northern_yuan_boss_scope = {
            add_casus_belli = {
                type = casus_belli:cb_subjugation
                target = root
                years = 400
            }
            add_opinion = {
                target = root
                modifier = opinion_northern_yuan_reform_opposed
            }
        }
        
        international_organization:northern_yuan = {
            remove_country_from_international_organization = root
        }
        add_opinion = {
            target = scope:northern_yuan_boss_scope
            modifier = opinion_northern_yuan_reform_opposed
        }
    }
}

northern_yuan.200 = { #Uriankhai Tumen Event
    type = country_event
    title = northern_yuan.200.title
    desc = northern_yuan.200.desc

    illustration_tags = {
        10 = regular
        10 = exterior
    }

    immediate = {
        event_illustration_poptype_effect = { foreground = pop_type:soldiers background = pop_type:soldiers }
        if = {
            limit = { c:KUH = { is_ai = no } }
            c:KUH = { save_scope_as = uriankhai_former }
        }
        else_if = {
            limit = { c:KVU = { is_ai = no } }
            c:KVU = { save_scope_as = uriankhai_former }
        }
        else_if = {
            limit = { c:TBU = { is_ai = no } }
            c:TBU = { save_scope_as = uriankhai_former }
        }
        else_if = {
            limit = { c:AUK = { is_ai = no } }
            c:AUK = { save_scope_as = uriankhai_former }
        }
        else = {
            c:KVU = { save_scope_as = uriankhai_former }
        }
    }

    option = {
        name = northern_yuan.200.a
        custom_tooltip = {
            text = northern_yuan.200.ctt
            scope:uriankhai_former = {
                every_country = {
                    limit = {
                        OR = {
                            tag = KUH
                            tag = KVU
                            tag = TBU
                            tag = AUK
                        }
                        NOT = { this = scope:uriankhai_former }
                        is_subject = no
                    }
                    scope:uriankhai_former = {
                        annex_country = {
                            country = prev
                            reason = Diplomatic
                        }
                    }
                }
                form_country = formable_country:URZ_f
            }
        }
        change_yeke_mongghul_ulus = { value = 5 }
    }
}
