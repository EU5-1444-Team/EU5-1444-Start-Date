namespace = io_unification

# Common member annexation event (reused via trigger_event_silently)
# scope:target_country must be saved by the triggering event
io_unification.1 = {
	type = country_event
	title = io_unification.1.title
	desc = io_unification.1.desc
	outcome = neutral

	illustration_tags = {
		10 = happy
		10 = interior
	}

	immediate = {
		event_illustration_estate_effect = { foreground = estate_type:nobles_estate background = estate_type:nobles_estate }
	}

	option = {
		name = io_unification.1.a

		if = {
			limit = {
				country_type = location
			}
			scope:target_country = {
				annex_country_diplomatically = { country = root }
			}
		}
	}
}

io_unification.10 = { # Khalkha unification
	type = country_event
	title = io_unification.10.title
	desc = io_unification.10.desc
	outcome = neutral

	illustration_tags = {
		10 = happy
		10 = interior
	}

	immediate = {
		event_illustration_estate_effect = { foreground = estate_type:nobles_estate background = estate_type:nobles_estate }
		save_scope_as = target_country
	}

	option = {
		name = io_unification.10.a
		form_country = formable_country:KHL_f
		add_reform = government_reform:eastern_steppe_unification_reform
		if = {
			limit = { exists = international_organization:khalkha_confederation }
			custom_tooltip = {
				text = io_unification.10.ctt
				if = {
					limit = { exists = scope:target_country }
					international_organization:khalkha_confederation = {
						every_international_organization_member = {
							limit = { NOT = { this = scope:target_country } }
							trigger_event_silently = { id = io_unification.1 }
						}
					}
				}
			}
			destroy_international_organization = { target = international_organization:khalkha_confederation }
		}
	}
}

# Bashkir tribal unification
io_unification.40 = {
    type = country_event
    title = io_unification.40.title
    desc = io_unification.40.desc
    outcome = neutral
    illustration_tags = {
        10 = happy
        10 = interior
    }
    immediate = {
        event_illustration_estate_effect = { foreground = estate_type:nobles_estate background = estate_type:nobles_estate }
        save_scope_as = target_country
        every_international_organization = {
            limit = {
                international_organization_type = international_organization_type:european_tribal_confederation
                leader_country = scope:target_country
            }
            save_scope_as = target_io
        }
    }
    option = {
        name = io_unification.40.a
        add_reform = government_reform:european_tribe_unification_reform
        if = {
            limit = { exists = scope:target_io }
            custom_tooltip = {
                text = io_unification.40.ctt
                scope:target_io = {
                    every_international_organization_member = {
                        limit = {
							NOT = { this = scope:target_country }
							NOT = {
                                has_special_status_in_international_organization = {
                                    type = special_status:tatar_overlord
                                    international_organization = scope:target_io
                                }
                            }
						}
                        trigger_event_silently = { id = io_unification.1 }
                    }
                }
            }
            destroy_international_organization = { target = scope:target_io }
        }
    }
}

# Jurchen unification
io_unification.90 = {
	type = country_event
	title = io_unification.90.title
	desc = io_unification.90.desc
	outcome = neutral

	illustration_tags = {
		10 = happy
		10 = interior
	}

	immediate = {
		event_illustration_estate_effect = { foreground = estate_type:nobles_estate background = estate_type:nobles_estate }
		save_scope_as = target_country
		if = {
			limit = {
				OR = {
					tag = NAN
					tag = UDE
					tag = WRK
					tag = SFN
					tag = YER
				}
			}
			form_country = formable_country:YER_f
		}
		else_if = {
			limit = {
				culture ?= culture:haixi_culture
			}
			form_country = formable_country:HXI_f
		}
		else = {
			form_country = formable_country:JNZ_f
		}
	}

	option = {
		name = io_unification.90.a
		trigger_event_non_silently = jurchen_confederation_events.17
	}
}
