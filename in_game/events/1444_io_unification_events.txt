namespace = io_unification

# Common member annexation event (reused via trigger_event_silently)
# scope:unification_leader must be saved by the triggering event
io_unification.1 = {
	type = country_event
	hidden = yes

	option = {
		name = io_unification.1.a
		scope:unification_leader = {
			annex_country = {
				country = root
				reason = Diplomatic
			}
		}
	}
}

# Khalkha unification
io_unification.10 = {
	type = country_event
	hidden = yes

	immediate = {
		save_scope_as = unification_leader
	}

	option = {
		name = io_unification.10.a
		international_organization:khalkha_confederation = {
			every_international_organization_member = {
				limit = { NOT = { this = root } }
				trigger_event_silently = { id = io_unification.1 }
			}
		}
		scope:unification_leader = {
			form_country = formable_country:KHL_f
			add_reform = government_reform:eastern_steppe_unification_reform
			if = {
				limit = { country_rank_level < 3 }
				set_country_rank_effect = {
					rank = country_rank:rank_kingdom
				}
			}
		}
		destroy_international_organization = { target = international_organization:khalkha_confederation }
	}
}
