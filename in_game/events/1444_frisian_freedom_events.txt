namespace = frisian_freedom_unification

frisian_freedom_unification.1 = {
	type = country_event
	hidden = yes

	immediate = {
		save_scope_as = unification_leader
	}

	option = {
		name = frisian_freedom_unification.1.a
		international_organization:frisian_freedom = {
			every_international_organization_member = {
				limit = {
					NOT = { this = root }
				}
				trigger_event_silently = { id = frisian_freedom_unification.2 }
			}
		}
		scope:unification_leader = {
			form_country = formable_country:FRI_f
			if = {
				limit = {
					country_rank_level < 3
				}
				set_country_rank_effect = {
					rank = country_rank:rank_kingdom
				}
			}
		}
		destroy_international_organization = { target = international_organization:frisian_freedom }
	}
}

frisian_freedom_unification.2 = {
	type = country_event
	hidden = yes

	option = {
		name = frisian_freedom_unification.2.a
		scope:unification_leader = {
			annex_country = {
				country = root
				reason = Diplomatic
			}
		}
	}
}
