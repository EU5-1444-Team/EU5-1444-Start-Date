namespace = flavor_frisia
flavor_frisia.1 = {
	type = country_event
	title = flavor_frisia.1.t
	desc = flavor_frisia.1.d
	fire_only_once = yes
	dynamic_historical_event = {
		tag = EFR
		from = 1465.1.1
		to = 1470.1.1
		monthly_chance = 100
	}
	illustration_tags = {
		10 = regular
		10 = interior
	}
	immediate = {
		event_illustration_estate_effect = {
			foreground = estate_type:nobles_estate
			background = estate_type:nobles_estate
		}
	}
	option = {
		name = flavor_frisia.1.a
		change_government_type = government_type:monarchy
		if = {
			limit = {
				country_exists = c:EMD
			}
			c:EMD = {
				make_subject_of = {
					target = root
					type = subject_type:vassal
				}
			}
		}
		international_organization:hre = {
			add_country_to_international_organization = root
		}
		ai_chance = {
			base = 30
		}
	}
	option = {
		name = flavor_frisia.1.b
		ai_chance = {
			base = 70
		}
	}
}

flavor_frisia.2 = {
	type = country_event
	title = flavor_frisia.2.t
	desc = flavor_frisia.2.d
	fire_only_once = yes
	illustration_tags = {
		10 = angry
		10 = exterior
	}
	trigger = {
		tag = HOL
		location:hoorn = {
			owner = root
			any_pop = {
				culture = culture:frisian
			}
		}
		unique_international_organization_type_exists = international_organization_type:frisian_freedom
		international_organization:frisian_freedom = {
			international_organization_has_leader = yes
		}
	}
	dynamic_historical_event = {
		tag = HOL
		from = 1444.11.11
		to = 1500.1.1
		monthly_chance = 2
	}
	immediate = {
		event_illustration_estate_effect = {
			foreground = estate_type:nobles_estate
			background = estate_type:nobles_estate
		}
	}
	option = {
		name = flavor_frisia.2.a
		# Autonomy - make vassal
		create_country_from_cores_in_our_locations = c:WFR
		c:WFR = {
			make_subject_of = {
				target = root
				type = subject_type:vassal
			}
		}
		international_organization:frisian_freedom = {
			add_country_to_international_organization = c:WFR
		}
		ai_chance = {
			base = 40
		}
	}
	option = {
		name = flavor_frisia.2.b
		# Complete independence
		create_country_from_cores_in_our_locations = c:WFR
		international_organization:frisian_freedom = {
			add_country_to_international_organization = c:WFR
		}
		ai_chance = {
			base = 10
		}
	}
	option = {
		name = flavor_frisia.2.c
		# Independence war
		create_country_from_cores_in_our_locations = c:WFR
		c:WFR = {
			make_subject_of = {
				target = root
				type = subject_type:vassal
			}
			declare_war_with_cb = {
				target = root
				type = casus_belli:cb_independence_war
			}
			"war_with_country(root)" = {
				save_scope_as = new_wfr_independence_war
			}
		}
		international_organization:frisian_freedom = {
			add_country_to_international_organization = c:WFR
		}
		if = {
			limit = {
				exists = scope:new_wfr_independence_war
			}
			international_organization:frisian_freedom = {
				every_international_organization_member = {
					join_war_with = {
						target = c:WFR
						war = scope:new_wfr_independence_war
					}
				}
			}
		}
		ai_chance = {
			base = 50
		}
	}
}
