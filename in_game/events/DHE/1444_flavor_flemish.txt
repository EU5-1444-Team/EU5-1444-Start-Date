flemish_flavor.1 = {
	type = country_event

	fire_only_once = yes
	hidden = yes

	dynamic_historical_event = {
		tag = FLA
		tag = BRB
		tag = BEL
		from = 1700.1.1
		to = 1836.1.1
		monthly_chance = 10
	}

	trigger = {
		current_age = age_6_revolutions
		OR = {
			culture = culture:flemish
			tag = FLA
			tag = BRB
			tag = BEL
		}
		any_owned_location = {
			area = area:flanders_area
			any_pop = {
				culture = culture:low_franconian
			}
		}
	}

	immediate = {
		every_owned_location = {
			limit = {
				area = area:flanders_area
			}
			every_pop = {
				limit = {
					culture = culture:low_franconian
				}
				change_pop_culture = culture:flemish
			}
		}
	}
}
