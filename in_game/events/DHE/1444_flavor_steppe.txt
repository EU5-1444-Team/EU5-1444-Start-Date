flavor_steppe.3 = { # Rise of Astrakhan
    type = country_event
    title = flavor_steppe.3.title
    desc = flavor_steppe.3.desc
    fire_only_once = yes

    dynamic_historical_event = {
        tag = GTH
        from = 1455.1.1
        to = 1850.1.1
        monthly_chance = 100
    }
    trigger = {
        has_or_had_tag = GTH
        has_global_variable = split_horde
        country_exists = c:ATR
    }
    immediate = {
        character:gth_mahmud_bin_kuchuk = { save_scope_as = target_character2 }
        c:ATR = { save_scope_as = target_country }
        scope:target_country = {
            set_new_ruler = scope:target_character2
            change_government_type = government_type:steppe_horde
        }
        random_current_war = {
            limit = {
                OR = {
                    AND = {
                        attacker_leader = root
                        defender_leader = scope:target_country
                    }
                    AND = {
                        attacker_leader = scope:target_country
                        defender_leader = root
                    }
                }
            }
            save_scope_as = target_war
        }
    }

    option = {
        name = flavor_steppe.3.a #Go to War
        ai_chance = { factor = 0.1 }
    }
    option = {
        name = flavor_steppe.3.b #Accept the breakaway
        ai_chance = { factor = 0.9 }
        if = {
            limit = { exists = scope:target_war }
            white_peace = scope:target_war
        }
        add_prestige = prestige_mild_penalty
    }
    option = {
        name = flavor_steppe.3.c #Play as Astrakhan
        custom_tooltip = flavor_steppe_3_c_ctt
