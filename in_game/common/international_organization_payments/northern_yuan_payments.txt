yuan_tribute = {
    get_payer_list = {
        every_international_organization_member = {
            limit = {
                NOT = {
                    has_special_status_in_international_organization = {
                        type = special_status:protector_general
                        international_organization = root
                    }
                }
            }
            add_to_local_variable_list = {
                name = payers
                target = this
            }
        }
    }
    get_payee_list = {
        every_country_with_special_status_of_type = {
            type = protector_general
            add_to_local_variable_list = {
                name = payees
                target = this
            }
        }
    }
    price = yuan_tribute_price
    uses_maintenance = yes
    maintenance_modifier = {
        potential_trigger = { "payment_maintenance(scope:recipient|yuan_tribute)" < 0.5 }
        scale = {
            value = 1
            subtract = {
                value = "payment_maintenance(scope:recipient|yuan_tribute)"
                multiply = 2
            }
        }
        monthly_prestige = -0.02
    }
    maintenance_modifier = {
        potential_trigger = { "payment_maintenance(scope:recipient|yuan_tribute)" > 0.5 }
        scale = {
            value = "payment_maintenance(scope:recipient|yuan_tribute)"
            subtract = 0.5
            multiply = 2
        }
        monthly_prestige = 0.01
        legitimacy = 0.5
    }
    price_multiplier = {
        every_international_organization_member = {
            limit = {
                NOT = { is_leader_of_international_organization = root }
            }
            add = monthly_income_trade_and_tax
        }
        multiply = 0.0001
    }
    proportion_for_payer = {
        add = {
            desc = "yuan_tribute"
            value = monthly_income_trade_and_tax
        }
    }
    proportion_for_payee = {
        add = {
            desc = "yuan_tribute"
            value = 1
        }
    }
}
