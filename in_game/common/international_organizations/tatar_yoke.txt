tatar_yoke = {
	has_target = no
	unique = yes
	expel_members_who_are_targets_of_other_members = no

	has_leader_country = yes
	leader_title_key = "TATAR_YOKE_LEADER"
	leader_type = character
	leader = {
		leader_country ?= {
			ruler ?= {
				add_to_list = leaders
			}
		}
	}
	leader_change_trigger_type = none

	diplomatic_capacity_cost = {
		add = {
			desc = "UPKEEP_BASE_VALUE"
			value = 1
		}
	}

	can_declare_war = {
		always = yes
	}
	
	non_leader_modifier = {
		global_monthly_development = -0.001
	}
	
	leader_modifier = {
		monthly_tribal_cohesion = 0.1
		monthly_prestige = 0.1
	}
	
	create_visible_trigger = {
		always = no
	}

	invite_visible_trigger = {
		always = yes
	}

	can_lead_trigger = {
		has_ruler = yes
		ruler ?= {
			is_adult = yes
		}
	}
	
	can_join_trigger = {
		always = no
	}
	
	can_leave_trigger = {
		always = no
	}
	
	auto_leave_trigger = {
		
	}

	auto_disband_trigger = {
		total_members = 1
	}

	on_joined = {
	}

	on_left = {
	}

	monthly_effect = {
		# Block 1 - Horde Leadership Transfer
		if = {
			limit = {
				any_international_organization_member = {
					NOT = { is_leader_of_international_organization = root }
					government_type = government_type:steppe_horde
					great_power_score > root.leader_country.great_power_score
				}
			}
			ordered_international_organization_member = {
				order_by = great_power_score
				max = 1
				check_range_bounds = no
				limit = {
					NOT = { is_leader_of_international_organization = root }
					government_type = government_type:steppe_horde
				}
				root = {
					set_leader_country = prev
				}
			}
		}
		
		# Block 2 - Tax Collector Reassignment
		if = {
			limit = {
				any_international_organization_member = {
					has_special_status_in_international_organization = {
						type = special_status:tatar_tax_collector
						international_organization = root
					}
				}
			}
			every_international_organization_member = {
				limit = {
					has_special_status_in_international_organization = {
						type = special_status:tatar_tax_collector
						international_organization = root
					}
				}
				root = {
					set_variable = {
						name = current_tax_collector_score
						value = prev.great_power_score
					}
				}
			}
		}
		if = {
			limit = {
				any_international_organization_member = {
					NOT = {
						has_special_status_in_international_organization = {
							type = special_status:tatar_tax_collector
							international_organization = root
						}
					}
					culture = { has_culture_group = culture_group:russian_group }
					great_power_score > root.var:current_tax_collector_score
				}
			}
			ordered_international_organization_member = {
				order_by = great_power_score
				max = 1
				check_range_bounds = no
				limit = {
					culture = { has_culture_group = culture_group:russian_group }
					NOT = {
						has_special_status_in_international_organization = {
							type = special_status:tatar_tax_collector
							international_organization = root
						}
					}
				}
				save_scope_as = new_tax_collector
			}
			every_international_organization_member = {
				limit = {
					has_special_status_in_international_organization = {
						type = special_status:tatar_tax_collector
						international_organization = root
					}
				}
				root = {
					international_organization_remove_special_status = {
						type = special_status:tatar_tax_collector
						country = prev
					}
				}
			}
			root = {
				international_organization_add_special_status = {
					type = special_status:tatar_tax_collector
					country = scope:new_tax_collector
				}
			}
		}
		
		# Block 3 - Great Ugra Stand Dissolution Detection
		if = {
			limit = {
				any_international_organization_member = {
					NOT = { is_leader_of_international_organization = root }
					culture = { has_culture_group = culture_group:russian_group }
					great_power_score > {
						value = root.leader_country.great_power_score
						multiply = 1.5
					}
				}
			}
			ordered_international_organization_member = {
				order_by = great_power_score
				max = 1
				check_range_bounds = no
				limit = {
					culture = { has_culture_group = culture_group:russian_group }
					NOT = { is_leader_of_international_organization = root }
				}
				root = {
					set_variable = {
						name = ugra_rus_prince
						value = prev
					}
				}
			}
			root = {
				set_variable = {
					name = ugra_horde_leader
					value = root.leader_country
				}
			}
			every_international_organization_member = {
				trigger_event_non_silently = tatar_yoke_events.1
			}
		}
	}

	international_organization_modifier = {
		cannot_declare_no_cb_wars_on_members = yes
	}
	
	ai_desire_to_join = {
	}
	
	ai_desire_to_allow_new_member = {
	}

	payments_implemented = {
		tatar_yoke_contribution
		tatar_yoke_leader_payments
	}

	special_statuses_implemented = {
		tatar_overlord
		tatar_tax_collector
		tatar_horde_member
	}
}
