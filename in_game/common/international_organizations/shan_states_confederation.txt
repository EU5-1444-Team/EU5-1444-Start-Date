shan_states_confederation = {
	has_target = yes
	unique = no
	expel_members_who_are_targets_of_other_members = no
	expel_members_who_are_attackers_at_war_with_other_members = yes
	expel_members_who_are_defenders_at_war_with_other_members = yes
	custom_name = shan_states_confederation_name
	show_on_diplomatic_map = yes
	has_leader_country = yes
	leader_type = none
	leader_change_method = none
	leader_change_trigger_type = none
	months_between_leader_changes = 48
	disband_if_no_leader = no
	show_strength_comparison_with_target = yes
	declare_war_on_target_casus_belli = cb_shan_confederation_raid
	gives_food_access_to_members = yes
	leader_color = define:NMapColors|INTERNATIONAL_ORGANIZATION_LEADER_COLOR
	target_color = define:NMapColors|INTERNATIONAL_ORGANIZATION_TARGET_COLOR
	join_offensive_wars_always = {
		scope:target ?= {
			this = root.international_organization_target
		}
	}
	join_defensive_wars_always = {
		scope:target ?= {
			this = root.international_organization_target
		}
	}
	create_visible_trigger = {
		always = no
	}
	invite_visible_trigger = {
		always = yes
	}
	can_declare_war = {
		always = yes
	}
	can_initiate_policy_votes = {
		is_member_of_international_organization = scope:recipient
	}
	laws = {
		mutual_defense_law = assured_defense_policy
		mutual_offense_law = possible_offense_policy
	}
	modifier = {
	}
	target_modifier = {
		monthly_complacency = -0.05
	}
	potential_target_trigger = {
		this = c:AVV
	}
	can_target_trigger = {
		this = c:AVV
	}
	can_join_trigger = {
		not = {
			is_member_of_international_organization_of_type = {
				type = shan_states_confederation
				target = scope:target
			}
		}
		OR = {
			capital ?= {
				region = region:indochina_region
			}
			scope:recipient ?= {
				international_organization_has_policy = policy:shan_diplomatic_expansion_policy
			}
		}
		culture = {
			has_culture_group = culture_group:shan_group
		}
	}
	can_leave_trigger = {
	}
	auto_leave_trigger = {
		NOT = {
			culture = {
				has_culture_group = culture_group:shan_group
			}
		}
	}
	auto_disband_trigger = {
		total_members < 2
	}
	on_joined = {
		add_casus_belli = {
			target = scope:target
			type = casus_belli:cb_shan_confederation_raid
		}
	}
	on_left = {
		remove_casus_belli = {
			target = scope:target
			type = casus_belli:cb_shan_confederation_raid
		}
	}
	has_military_access = {
		scope:recipient ?= {
			is_member_of_international_organization = root
		}
	}
	variables = {
	}
	ai_desire_to_join = {
		add = {
			desc = "AT_WAR_TRIGGER"
			if = {
				limit = {
					is_at_war_with = scope:target
				}
				value = 20
			}
		}
		add = {
			desc = "game_concept_antagonism"
			scope:target = {
				value = prev_antagonism_towards_this
			}
		}
		if = {
			limit = {
				exists = scope:recipient
			}
			scope:recipient ?= {
				add = {
					desc = "CURRENT_SIZE_COMPARED_TO_TARGET"
					value = "organization_strength_relative_to_country(scope:target|yes)"
					add = -5
					multiply = -25
				}
			}
		}
		add = {
			desc = "DISTANCE_TO_TARGET"
			scope:target = {
				value = "border_distance_to(root)"
				multiply = -0.5
			}
		}
		if = {
			limit = {
				scope:target = {
					"border_distance_to(root)" > 300
				}
			}
			add = -1000
		}
		add = {
			desc = "OPINION"
			value = "opinion(scope:target)"
			multiply = -0.5
			max = 0
		}
	}
	ai_desire_to_allow_new_member = {
		add = {
			desc = "WE_WANT_THE_ORGANIZATION_TO_GROW"
			value = 20
		}
	}
	ai_issue_voting_bias = {
		add = {
			desc = "INERTIA"
			value = -1
		}
	}
}
