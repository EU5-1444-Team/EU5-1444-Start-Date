golden_horde = {
	has_target = no
	unique = yes
	expel_members_who_are_targets_of_other_members = no
	disband_if_no_leader = no
	show_as_overlord_on_map_trigger = {
		scope:recipient = {
			international_organization_has_leader = yes
		}
	}
	
	has_leader_country = yes
	leader_title_key = "GOLDEN_HORDE_LEADER"
	leader_type = character
	leader = {
		leader_country ?= {
			if = {
				limit = { 
					has_regent = yes
					has_heir = yes
				}
				heir = {
					add_to_list = leaders
				}
			}
			else = {
				ruler ?= {
					add_to_list = leaders
				}
			}
		}
	}
	leader_change_trigger_type = none
	leader_change_method = none
	can_lead_trigger = {
	}
	only_leader_country_joins_defensive_wars = yes
	join_defensive_wars_auto_call = {
		scope:target ?= {
			NOT = {
				is_member_of_international_organization = root
			}
		}
	}
	
	can_declare_war = {
		always = yes
	}
	gives_military_access_to_all_when_at_war = yes
	modifier = {
		block_from_change_to_empire_rank = yes
	}
	
	leader_modifier = {
		great_power_score_exempt_from_forfeit = 5
	}
	
	create_visible_trigger = {
		any_owned_location = { 
			region = region:steppes_region
		}
		primary_culture = culture:cuman_culture
	}
	invite_visible_trigger = {
		any_owned_location = { 
			region = region:steppes_region
		}
		primary_culture = culture:cuman_culture
	}
	can_join_trigger = {
       	primary_culture = culture:cuman_culture
    }
	
	can_leave_trigger = {
		always = no
	}
	
	auto_leave_trigger = {
		always = no
	}
	
	auto_disband_trigger = {
		total_members = 0
		any_international_organization_member = {
			NOT = { has_variable = forced_to_renounce_khan_claim }
			NOT = {
				has_special_status_in_international_organization = {
					type = special_status:golden_horde_claimant
					international_organization = scope:recipient
				}
			}
			NOT = { is_leader_of_international_organization = international_organization:golden_horde }
			NOR = {
                ruler_or_heir_if_regent ?= { 
                    OR = {
                        is_patrilineal_descendant_of = dynasty:giray_dynasty
                        is_patrilineal_descendant_of = dynasty:jochi_dynasty
                    }
                }
                any_character = { 
                    OR = {
                        is_patrilineal_descendant_of = dynasty:giray_dynasty
                        is_patrilineal_descendant_of = dynasty:jochi_dynasty
                    }
                }
            }
		}
	}
	on_joined = {
	}
	on_left = {
	}
	
	monthly_effect = {
		if = {
			limit = { "num_countries_with_special_status(special_status:golden_horde_claimant)" = 1 }
			every_international_organization_member = {
				limit = {
					has_special_status_in_international_organization = {
						type = special_status:golden_horde_claimant
						international_organization = root
					}
				}
				root = {
					set_leader_country = prev
					international_organization_remove_special_status = {
						type = special_status:golden_horde_claimant
						country = prev
					}
				}
			}
		}
	}
	
	variables = {
	}
	special_statuses_implemented = {
		golden_horde_claimant
	}
	ai_desire_to_join = {
	}
	
	ai_desire_to_allow_new_member = {
		add = {
			desc = "WE_WANT_THE_ORGANIZATION_TO_GROW"
			value = 20
		}
	}
}
