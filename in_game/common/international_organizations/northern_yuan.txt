northern_yuan = {
    has_target = no
    unique = yes
    expel_members_who_are_targets_of_other_members = no
    disband_if_no_leader = no

    show_as_overlord_on_map_trigger = {
        always = yes
    }

    show_on_diplomatic_map = yes

    has_leader_country = yes
    leader_title_key = "NORTHERN_YUAN_LEADER"
    leader_type = character
    leader = {
        leader_country ?= {
            if = {
                limit = {
                    has_regent = yes
                    has_heir = yes
                }
                heir ?= {
                    add_to_list = leaders
                }
            }
            else_if = {
                limit = {
                    has_ruler = yes
                    ruler ?= { is_alive = yes }
                }
                ruler ?= {
                    add_to_list = leaders
                }
            }
        }
    }
    leader_change_trigger_type = none
    leader_change_method = none

    can_lead_trigger = {
    }
    payments_implemented = {
        yuan_tribute
    }

    can_declare_war = {
        trigger_if = {
            limit = {
                scope:attacker = { is_member_of_international_organization = international_organization:northern_yuan }
                scope:defender = { is_member_of_international_organization = international_organization:northern_yuan }
            }
            OR = {
                scope:attacker = { has_casus_belli_of_type_on = { type = casus_belli:cb_seize_yuan target = scope:defender } }
                scope:attacker = { has_casus_belli_of_type_on = { type = casus_belli:cb_seize_protector_general target = scope:defender } }
                if = {
                    limit = {
                        NOT = {
                            OR = {
                                has_special_status_in_international_organization = { type = special_status:protector_general international_organization = international_organization:northern_yuan }
                                international_organization_has_policy = policy:administrative_integration_policy
                                international_organization_has_policy = policy:military_subjugation_policy
                            }
                        }
                    }
                    scope:attacker = { has_casus_belli_of_type_on = { type = casus_belli:cb_northern_yuan_raid target = scope:defender } }
                }
            }
        }
    }

    land_ownership_rule = northern_yuan_land_ownership
    
    join_defensive_wars_auto_call = {
        scope:target ?= {
            NOT = {
                is_member_of_international_organization = root
            }
        }
    }

    gives_military_access_to_all_when_at_war = yes

    can_initiate_policy_votes = {
        is_member_of_international_organization = international_organization:northern_yuan
        has_special_status_in_international_organization = {
            type = special_status:protector_general
            international_organization = international_organization:northern_yuan
        }
    }

    modifier = {
        block_from_change_to_empire_rank = yes
    }

    leader_modifier = {
        great_power_score_exempt_from_forfeit = 20
    }

    create_visible_trigger = {
        always = no
    }
    invite_visible_trigger = {
        always = yes
    }

    can_join_trigger = {
        culture = {
            has_culture_group = culture_group:mongolian_group
        }
    }

    can_leave_trigger = {
        always = no
    }

    auto_leave_trigger = {
        always = no
    }

    auto_disband_trigger = {
        total_members < 1
    }

    on_joined = {}
    on_left = {}

    monthly_effect = {
        hidden_effect = {
            if = {
                limit = {
                    international_organization_has_leader = no
                }
                ordered_international_organization_member = {
                    order_by = great_power_score
                    max = 1
                    check_range_bounds = no
                    limit = {
                        ruler_or_heir_if_regent ?= {
                            OR = {
                                is_patrilineal_descendant_of = dynasty:borjigin_dynasty
                                is_patrilineal_descendant_of = dynasty:jochi_dynasty
                                is_patrilineal_descendant_of = dynasty:giray_dynasty
                                is_patrilineal_descendant_of = dynasty:chagatayid_dynasty
                                is_patrilineal_descendant_of = dynasty:khasar_dynasty
                            }
                        }
                    }
                    root = {
                        set_leader_country = prev
                    }
                }
                if = {
                    limit = {
                        international_organization_has_leader = no
                    }
                    ordered_international_organization_member = {
                        order_by = great_power_score
                        max = 1
                        check_range_bounds = no
                        root = {
                            set_leader_country = prev
                        }
                    }
                }
            }
        }
        # Force-rejoin Mongol vassals of the Taishi
        hidden_effect = {
            every_country = {
                limit = {
                    culture = { has_culture_group = culture_group:mongolian_group }
                    NOT = { is_member_of_international_organization = international_organization:northern_yuan }
                    is_subject = yes
                    top_overlord = {
                        has_special_status_in_international_organization = {
                            type = special_status:protector_general
                            international_organization = international_organization:northern_yuan
                        }
                    }
                }
                international_organization:northern_yuan = {
                    add_country_to_international_organization = prev
                }
            }
        }
    }

    variables = {
        yeke_mongghul_ulus = {
            format = "YEKE_MONGGHUL_ULUS_DISPLAY"
            change_format = "VARIABLE_CHANGE_FORMAT"
            min = 0
            max = 100
            monthly_change = {
                # Base decay
                add = {
                    desc = "BASE_DECAY"
                    value = -0.1
                }
                
                # Taishi prestige
                add = {
                    desc = "TAISHI_PRESTIGE"
                    value = 0
                    every_country_with_special_status_of_type = {
                        type = protector_general
                        add = modifier:monthly_prestige
                        multiply = 0.5
                    }
                }
                
                # Taishi military ruler stat
                add = {
                    desc = "TAISHI_MARTIAL"
                    value = 0
                    every_country_with_special_status_of_type = {
                        type = protector_general
                        add = {
                            value = ruler_or_heir_if_regent.mil
                            subtract = 50
                            multiply = 0.004
                        }
                    }
                }

                # Taishi Admin ruler stat
                add = {
                    desc = "TAISHI_ADMIN"
                    value = 0
                    every_country_with_special_status_of_type = {
                        type = protector_general
                        add = {
                            value = ruler_or_heir_if_regent.adm
                            subtract = 50
                            multiply = 0.001
                        }
                    }
                }
                
                # Taishi tribal cohesion
                add = {
                    desc = "TAISHI_COHESION"
                    value = 0
                    every_country_with_special_status_of_type = {
                        type = protector_general
                        add = monthly_tribal_unity
                        multiply = 0.2
                    }
                }
                
                # Internal peace bonus
                add = {
                    desc = "INTERNAL_PEACE"
                    if = {
                        limit = {
                            NOT = {
                                any_international_organization_member = {
                                    at_war = yes
                                    any_current_war = {
                                        any_war_participant = {
                                            is_member_of_international_organization = root
                                            NOT = { this = prev }
                                        }
                                    }
                                }
                            }
                        }
                        value = 0.05
                    }
                }
                
                # Internal wars penalty
                add = {
                    desc = "INTERNAL_WARS"
                    every_international_organization_member = {
                        limit = {
                            at_war = yes
                            any_current_war = {
                                any_war_participant = {
                                    is_member_of_international_organization = root
                                    NOT = { this = prev }
                                }
                            }
                        }
                        add = -0.5
                    }
                }
                
                # Members subjugated by outsiders
                add = {
                    desc = "MEMBERS_SUBJUGATED_BY_OUTSIDERS"
                    every_international_organization_member = {
                        limit = {
                            is_subject = yes
                            NOT = { is_subject_type = tributary }
                            top_overlord ?= {
                                NOT = { is_member_of_international_organization = root }
                            }
                        }
                        add = -0.03
                    }
                }
                
                # Active law tier bonus — tiers 1-9 each contribute
                add = {
                    desc = "REFORM_PROGRESS_TIER_1"
                    if = {
                        limit = {
                            OR = {
                                international_organization_has_policy = policy:formal_kurultai_policy
                                international_organization_has_policy = policy:loose_acknowledgment_policy
                            }
                        }
                        value = 0.05
                    }
                }
                add = {
                    desc = "REFORM_PROGRESS_TIER_2"
                    if = {
                        limit = {
                            OR = {
                                international_organization_has_policy = policy:standardized_tribute_policy
                                international_organization_has_policy = policy:military_tribute_policy
                            }
                        }
                        value = 0.05
                    }
                }
                add = {
                    desc = "REFORM_PROGRESS_TIER_3"
                    if = {
                        limit = {
                            OR = {
                                international_organization_has_policy = policy:yuan_arbitration_policy
                                international_organization_has_policy = policy:traditional_territories_policy
                            }
                        }
                        value = 0.05
                    }
                }
                add = {
                    desc = "REFORM_PROGRESS_TIER_4"
                    if = {
                        limit = {
                            OR = {
                                international_organization_has_policy = policy:standardized_tumen_policy
                                international_organization_has_policy = policy:yuan_offensive_call_policy
                            }
                        }
                        value = 0.05
                    }
                }
                add = {
                    desc = "REFORM_PROGRESS_TIER_5"
                    if = {
                        limit = {
                            OR = {
                                international_organization_has_policy = policy:sinicized_court_policy
                                international_organization_has_policy = policy:traditional_mongol_court_policy
                            }
                        }
                        value = 0.05
                    }
                }
                add = {
                    desc = "REFORM_PROGRESS_TIER_6"
                    if = {
                        limit = {
                            OR = {
                                international_organization_has_policy = policy:administrative_integration_policy
                                international_organization_has_policy = policy:military_subjugation_policy
                            }
                        }
                        value = 0.05
                    }
                }
                add = {
                    desc = "REFORM_PROGRESS_TIER_7"
                    if = {
                        limit = {
                            OR = {
                                international_organization_has_policy = policy:yuan_imperial_authority_policy
                                international_organization_has_policy = policy:mongol_imperial_authority_policy
                            }
                        }
                        value = 0.1
                    }
                }
                
                # No leader penalty
                if = {
                    limit = { international_organization_has_leader = no }
                    add = {
                        desc = "NO_LEADER"
                        value = -0.5
                    }
                }
                add = {
                    desc = "YUAN_LEGITIMACY_TENSION"
                    if = {
                        limit = { international_organization_has_leader = yes }
                        value = -0.05
                    }
                }
                # Non-Genghisid Taishi penalty
                add = {
                    desc = "NON_GENGHISID_TAISHI"
                    every_country_with_special_status_of_type = {
                        type = protector_general
                        limit = {
                            NOT = {
                                ruler_or_heir_if_regent ?= {
                                    OR = {
                                        is_patrilineal_descendant_of = dynasty:borjigin_dynasty
                                        is_patrilineal_descendant_of = dynasty:jochi_dynasty
                                        is_patrilineal_descendant_of = dynasty:giray_dynasty
                                        is_patrilineal_descendant_of = dynasty:chagatayid_dynasty
                                        is_patrilineal_descendant_of = dynasty:khasar_dynasty
                                    }
                                }
                            }
                        }
                        add = -0.05
                    }
                }
            }
        }
    }

    special_statuses_implemented = {
        yuan_holder
        protector_general
    }

    ai_desire_to_join = {
        add = {
            desc = "SHARED_MONGOL_HERITAGE"
            value = 50 
        }
    }

    ai_desire_to_allow_new_member = {
        add = {
            desc = "WE_WANT_THE_ORGANIZATION_TO_GROW"
            value = 20
        }
    }
}
