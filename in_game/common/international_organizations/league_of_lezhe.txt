league_of_lezhe = {
	has_target = no
	unique = yes
	show_on_diplomatic_map = yes
	has_leader_country = yes
	subject_limited = no
	expel_members_who_are_targets_of_other_members = no
	expel_members_who_target_the_leader = no
	expel_members_who_are_attackers_at_war_with_other_members = no
	expel_members_who_are_defenders_at_war_with_other_members = no
	leader_title_key = "LEAGUE_OF_LEZHE_LEADER"
	leader_type = character
	leader = {
		leader_country ?= {
			if = {
				limit = {
					has_regent = yes
					has_heir = yes
				}
				heir ?= {
					add_to_list = leaders
				}
			}
			else_if = {
				limit = {
					has_ruler = yes
					ruler ?= {
						is_alive = yes
					}
				}
				ruler ?= {
					add_to_list = leaders
				}
			}
		}
	}
	leader_change_trigger_type = rulerchange
	leader_change_method = vote
	leadership_election_resolution = league_of_lezhe_election
	leader_score = {
		value = great_power_score
	}
	disband_if_no_leader = no
	create_visible_trigger = {
		always = no
	}
	invite_visible_trigger = {
		always = yes
	}
	join_visible_trigger = {
		tag != VEN
	}
	join_enabled_trigger = {
		tag != VEN
	}
	can_lead_trigger = {
		NOT = {
			has_special_status_in_international_organization = {
				type = special_status:league_backer
				international_organization = scope:recipient
			}
		}
		NOT = {
			has_special_status_in_international_organization = {
				type = special_status:venetian_governorate
				international_organization = scope:recipient
			}
		}
		OR = {
			AND = {
				scope:recipient = {
					leader_country ?= root
				}
				heir ?= {
					is_adult = yes
				}
			}
			AND = {
				scope:recipient = {
					NOT = {
						leader_country ?= root
					}
				}
				ruler ?= {
					is_adult = yes
				}
			}
		}
	}
	can_declare_war = {
		always = yes
	}
	can_initiate_policy_votes = {
		is_member_of_international_organization = international_organization:league_of_lezhe
		is_leader_of_international_organization = international_organization:league_of_lezhe
	}
	land_ownership_rule = league_of_lezhe_land_ownership
	laws = {
		league_of_lezhe_defense_law = league_of_lezhe_defense_policy
		league_of_lezhe_offense_law = league_of_lezhe_no_offense_policy
		league_of_lezhe_venetian_governance_law = no_venetian_local_governors_policy
		league_of_lezhe_unification_law = preserve_league_of_lezhe_policy
	}
	special_statuses_implemented = {
		league_captain
		league_backer
		venetian_governorate
	}
	can_join_trigger = {
		trigger_if = {
			limit = {
				tag = VEN
			}
			always = yes
		}
		trigger_else = {
			is_rebel_country = no
			capital.area = area:albania_area
			international_organization:league_of_lezhe = {
				international_organization_has_leader = yes
			}
			opinion = {
				target = international_organization:league_of_lezhe.leader_country
				value >= 100
			}
		}
	}
	ai_desire_to_join = {
		if = {
			limit = {
				international_organization:league_of_lezhe = {
					international_organization_has_leader = yes
				}
			}
			add = {
				desc = "POLICY_BIAS_OPINION_OF_IO_LEADER"
				value = "opinion(international_organization:league_of_lezhe.leader_country)"
				multiply = 0.4
			}
		}
	}
	ai_desire_to_allow_new_member = {
		add = {
			desc = "WE_WANT_THE_ORGANIZATION_TO_GROW"
			value = 1000
		}
	}
	can_leave_trigger = {
		always = yes
	}
	auto_leave_trigger = {
		always = no
	}
	auto_disband_trigger = {
		any_international_organization_member = {
			count < 2
			NOT = {
				has_special_status_in_international_organization = {
					type = special_status:league_backer
					international_organization = root
				}
			}
		}
	}
	on_creation = {
		if = {
			limit = {
				country_exists = c:VEN
				c:VEN = {
					is_member_of_international_organization = root
				}
			}
			international_organization_add_special_status = {
				type = special_status:league_backer
				country = c:VEN
			}
		}
	}
	on_joined = {
		hidden_effect = {
			if = {
				limit = {
					is_subject_type = governorate
					overlord ?= {
						this = c:VEN
					}
				}
				international_organization:league_of_lezhe = {
					international_organization_add_special_status = {
						type = special_status:venetian_governorate
						country = root
					}
				}
			}
			if = {
				limit = {
					tag = VEN
				}
				international_organization:league_of_lezhe = {
					international_organization_add_special_status = {
						type = special_status:league_backer
						country = c:VEN
					}
				}
			}
			if = {
				limit = {
					NOT = {
						tag = VEN
					}
				}
				every_owned_location = {
					limit = {
						area = area:albania_area
					}
					international_organization:league_of_lezhe = {
						add_location_to_international_organization = prev
					}
				}
			}
		}
	}
	on_left = {
		hidden_effect = {
			if = {
				limit = {
					has_special_status_in_international_organization = {
						type = special_status:venetian_governorate
						international_organization = international_organization:league_of_lezhe
					}
				}
				international_organization:league_of_lezhe = {
					international_organization_remove_special_status = {
						type = special_status:venetian_governorate
						country = root
					}
				}
			}
			if = {
				limit = {
					tag = VEN
				}
				international_organization:league_of_lezhe = {
					international_organization_remove_special_status = {
						type = special_status:league_backer
						country = c:VEN
					}
				}
			}
		}
		every_owned_location = {
			limit = {
				is_owned_by_international_organization = international_organization:league_of_lezhe
			}
			international_organization:league_of_lezhe = {
				remove_location_from_international_organization = prev
			}
		}
	}
}
