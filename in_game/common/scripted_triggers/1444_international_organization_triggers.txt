has_active_main_branch = {
    international_organization_has_policy = policy:union_establish_seniority_policy
    trigger_if = {
        limit = { has_special_status_available = special_status:main_branch }
        any_country_with_special_status_of_type = {
            type = main_branch
            government_type = government_type:monarchy
        }
    }
    trigger_else = {
        has_special_status_available = special_status:main_branch
    }
}

country_is_main_branch = {
    international_organization_has_policy = policy:union_establish_seniority_policy
    country_has_special_status = {
        type = special_status:main_branch
        country = $country$
    }
}

northern_yuan_can_select_adjacent_reform_level = {
	$law$ = {
		save_temporary_scope_as = law_scope
	}
	$policy$ = {
		save_temporary_scope_as = policy_scope
	}
	trigger_if = {
		limit = {
			exists = scope:law_scope
			exists = scope:policy_scope
			international_organization_has_law = scope:law_scope
			exists = "law_policy(scope:law_scope)"
		}
		trigger_if = {
			limit = { "law_policy(scope:law_scope)" = { policy_level > policy_level_upper_level } }
			custom_description = {
				text = northern_yuan_can_select_adjacent_reform_level_low
				object = scope:law_scope
				value = policy_level_upper_level
				"law_policy(scope:law_scope)" = { policy_level <= policy_level_upper_level }
			}
		}
		trigger_else = {
			hidden_trigger = { always = yes }
		}
		trigger_if = {
			limit = { "law_policy(scope:law_scope)" = { policy_level < policy_level_lower_level } }
			custom_description = {
				text = northern_yuan_can_select_adjacent_reform_level_up
				object = scope:law_scope
				value = policy_level_lower_level
				"law_policy(scope:law_scope)" = { policy_level >= policy_level_lower_level }
			}
		}
		trigger_else = {
			hidden_trigger = { always = yes }
		}
	}
	trigger_else = { hidden_trigger = { always = yes } }
}
