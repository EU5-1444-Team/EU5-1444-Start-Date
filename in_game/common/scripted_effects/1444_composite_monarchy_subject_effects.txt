# Changes the composite_monarchy integration level
change_composite_integration = {
	hidden_effect = {
		if = {
			limit = { NOT = { has_variable = composite_integration } }
			set_variable = { name = composite_integration value = $value$ }
		}
		else = {
			change_variable = { name = composite_integration add = $value$ }
			if = {
				limit = { var:composite_integration < 0 }
				set_variable = { name = composite_integration value = 0 }
			}
			if = {
				limit = { var:composite_integration > 100 }
				set_variable = { name = composite_integration value = 100 }
			}
		}
	}
}

# Root = the international organization.
# Handles landing/unlanding for a composite monarchy member based on crown integration.
# Parameters:
#   tag = Country tag to manage (e.g., ATS)
#   overlord = Scope of the overlord country (e.g., scope:composite_monarchy_leader)
#   annex_threshold = Integration value to unland (e.g., 65)
#   release_threshold = Integration value to land (e.g., 40)
composite_monarchy_manage_subject = {
	# Unlanding: if location-type (landed) and integration >= threshold,
	# transfer provinces to overlord and switch to building-based (landless)
	if = {
		limit = {
			country_exists = $tag$
			c:$tag$ = {
				country_type = location
				is_subject = yes
				top_overlord = {
					this = $overlord$
				}
				$overlord$.var:composite_integration >= $annex_threshold$
			}
		}
		c:$tag$ = {
			every_owned_location = {
				change_location_owner = $overlord$
			}
			change_country_type = building
		}
	}
	# Landing: if building-type (landless) and integration < threshold,
	# switch to location-type and grant core provinces from overlord
	if = {
		limit = {
			country_exists = $tag$
			c:$tag$ = {
				country_type = building
				$overlord$.var:composite_integration < $release_threshold$
			}
		}
		c:$tag$ = {
			change_country_type = location
		}
		$overlord$ = {
			every_owned_location = {
				limit = {
					is_core_of = $tag$
				}
				change_location_owner = $tag$
			}
		}
	}
	# Cleanup: remove IO members that are not proper subjects of the overlord
	if = {
		limit = {
			country_exists = $tag$
			c:$tag$ = {
				is_member_of_international_organization = $overlord$
				NOT = {
					is_subject = yes
					top_overlord = {
						this = $overlord$
					}
				}
			}
		}
		c:$tag$ = {
			remove_from_international_organization = $overlord$
		}
	}
}
