frisian_freedom_land_ownership = {
	modifier = {
	}
	can_add_trigger = {
		always = yes
	}
	can_remove_trigger = {
		always = yes
	}
	can_add_location_trigger = {
		is_land = yes
		is_passable = yes
		owner ?= {
			is_member_of_international_organization = scope:recipient
		}
		OR = {
			area = area:friesland_area
			province = province:south_holland_province
			province = province:north_holland_province
			this = location:elsfleth
			this = location:ritzebuttel
			this = location:husum
			this = location:tonder
			any_neighbor_location = {
				is_owned_by_international_organization = scope:recipient
			}
		}
	}
	can_remove_location_trigger = {
		OR = {
			owner ?= {
				NOT = {
					is_member_of_international_organization = scope:recipient
				}
			}
			AND = {
				NOT = { area = area:friesland_area }
				NOT = { province = province:south_holland_province }
				NOT = { province = province:north_holland_province }
				NOT = { this = location:elsfleth }
				NOT = { this = location:ritzebuttel }
				NOT = { this = location:husum }
				NOT = { this = location:tonder }
				owner ?= {
					NOT = {
						is_leader_of_international_organization = scope:recipient
					}
				}
			}
		}
	}
	on_added = {
	}
	on_removed = {
	}
	ai_desire_to_add = {
	}
	allow_control_propagation = yes
}
