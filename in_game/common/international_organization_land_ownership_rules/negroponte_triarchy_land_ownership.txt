	modifier = {

		always = yes
	}
	can_remove_trigger = {
		always = yes
	}
	can_add_location_trigger = {
		is_land = yes
		is_passable = yes
		owner ?= {
			is_member_of_international_organization = scope:recipient
		}

			province = province:euboea_province
		}

		OR = {
			owner ?= {
				NOT = {
					is_member_of_international_organization = scope:recipient
				}
			}
			NOT = {
				province = province:euboea_province
			}
		}
	}

	}
	on_removed = {
	}
	ai_desire_to_add = {
	}
	allow_control_propagation = yes
