# Modded to not spawn on monthly tick due to Bohemia now starting as Hussite

hussite_wars = {
	monthly_spawn_chance = monthly_spawn_chance_unique
	hint_tag = hint_hussite_wars
	
	can_start = {	
		always = no
	}
	
	can_end = {
		end_reason = {
			trigger = { NOT = { country_exists = c:PAP } }
			desc = "papacy_dissolved"
		}
		end_reason = {
			trigger = { c:BOH ?= { religion = c:PAP.religion } }
			desc = "catholic_victory"
		}
		end_reason = {
			trigger = {
				AND = {
					situation:hussite_wars = { years_since_situation_start > 30 }
					c:BOH ?= {
						at_war = no
						NOT = { religion = religion:catholic }
					}
				}
			}
			desc = "hussite_victory"
		}
	}
	
	visible = {
		OR = {
			tag = BOH
			religion = religion:hussite
			religion = religion:catholic
		}
	}

	on_start = {
		religion:catholic = {
			every_country_in_religion = {
				limit = {
					OR = {
						is_member_of_international_organization = international_organization:hre
						is_neighbor_of = c:BOH
					}
					OR = {
						is_subject_of = c:BOH
						NOT = { is_allied_with = { target = c:BOH } }
					}
				}

				trigger_event_non_silently = hussite_wars.10
			}
		}

		set_variable = {
			name = heretics_eradicated_variable
			value = 0
		}

		set_variable = {
			name = heretics_converted_variable
			value = 0
		}

		c:BOH = { trigger_event_non_silently = hussite_wars.11 }
	}

	on_monthly = { 
		random_list = {
			99 = {}
			1 = { c:BOH = { trigger_event_non_silently = hussite_wars.8 } } 
		}

		random_list = {
			9 = {}
			1 = {
				trigger = {
					country_exists = international_organization:hre.leader_country
					international_organization:hre.leader_country = {
						NOT = { tag = BOH }
						religion = religion:catholic
						can_declare_war_on = c:BOH
						NOT = { has_truce_with = c:BOH }
						NOT = { is_at_war_with = c:BOH }
						at_war = no
						is_during_bankruptcy = no
						war_exhaustion_percentage < 0.2
					}
				}
				international_organization:hre.leader_country = { trigger_event_non_silently = hussite_wars.9 }
			}
		}
	}


	on_ended = {

		remove_variable = heretics_converted_variable
		remove_variable = heretics_eradicated_variable
		c:BOH ?= {
			every_other_country = {
				remove_opinion = {
					modifier = religious_disagreements_modifier
					target = c:BOH
				}
				remove_country_modifier = hussite_heresy_modifier
			}
			trigger_event_non_silently = hussite_wars.4
		}
		religion:hussite = {
			every_country_in_religion = {
				limit = { NOT = { tag = BOH } }
				trigger_event_non_silently = hussite_wars.4
			}
		}
		religion:catholic = {
			every_country_in_religion = {
				limit = {
					NOT = { tag = BOH }
					OR = {
						tag = PAP
						is_member_of_international_organization = international_organization:hre
						AND = {
							country_exists = c:BOH
							is_neighbor_of = c:BOH
						}
					}
				}
				trigger_event_non_silently = hussite_wars.4
			}
		}

		if = {
			limit = {	#Tracking for the achievement house_of_hus
				country_exists = c:BOH
				c:BOH.religion = religion:hussite
			}
			set_global_variable = {
				name = victorious_hussites_variable
				years = 1
			}
			set_global_variable = hussites_won_hussite_wars
		}
	}



	tooltip = {
		if = {
			limit = {	
				has_owner = yes
				owner = {
					OR = {
						AND = {
							is_subject_of = c:BOH
							subject_loyalty > 50
						}
						has_mutual_scripted_relation = {
							type = relation_type:alliance
							target = c:BOH
						}
					}
				}
			}
			custom_tooltip = "HUSSITE_WARS_SIDE_BOHEMIA"
		}
		else_if = {
			limit = {
				has_owner = yes
				owner = {
					OR = {
						AND = {
							is_subject_of = international_organization:hre.leader_country
							subject_loyalty > 50
						}
						has_mutual_scripted_relation = {
							type = relation_type:alliance
							target = international_organization:hre.leader_country
						}
					}
				}
			}
			custom_tooltip = "HUSSITE_WARS_SIDE_EMPEROR"
		}
		else_if = {
			limit = {
				has_owner = yes
				owner = international_organization:hre.leader_country
			}
			custom_tooltip = emperor_hw_tt
		}
		else_if = {
			limit = {
				has_owner = yes
				owner = c:BOH
			}
			custom_tooltip = boh_hw_tt
		}
		if = {
			limit = {
				has_owner = yes
				province = { has_province_modifier = recent_tribunal_modifier }
			}
			custom_tooltip = recent_tribunal_tt
		}
	}

	map_color = {
		if = {
			limit = {	
				has_owner = yes
				owner = {
					OR = {
						AND = {
							is_subject_or_below_of = c:BOH
							subject_loyalty > 50
						}
						has_mutual_scripted_relation = {
							type = relation_type:alliance
							target = c:BOH
						}
					}
				}
			}
			value = owner.country_color
		}
		else_if = {
			limit = {
				has_owner = yes
				owner = {
					OR = {
						AND = {
							is_subject_or_below_of = international_organization:hre.leader_country
							subject_loyalty > 50
						}
						has_mutual_scripted_relation = {
							type = relation_type:alliance
							target = international_organization:hre.leader_country
						}
					}
				}
			}
			value = owner.country_color
		}
		else_if = {
			limit = {
				has_owner = yes
				owner = international_organization:hre.leader_country
			}
			value = owner.country_color
		}
		else_if = {
			limit = {
				has_owner = yes
				owner = c:BOH
			}
			value = owner.country_color
		}
		else = {
			value = define:NMapColors|DEFAULT_COLOR
		}
	}

	secondary_map_color = {
		if = {
			limit = {
				has_owner = yes
				province = { has_province_modifier = recent_tribunal_modifier }
			}
			value = define:NMapColors|MAP_COLOR_LOW
		}
		if = {
			limit = {	
				has_owner = yes
				owner = {
					OR = {
						AND = {
							is_subject_or_below_of = c:BOH
							subject_loyalty > 50
						}
						has_mutual_scripted_relation = {
							type = relation_type:alliance
							target = c:BOH
						}
					}
				}
			}
			value = c:BOH.country_color
		}
		else_if = {
			limit = {
				has_owner = yes
				owner = {
					OR = {
						AND = {
							is_subject_or_below_of = international_organization:hre.leader_country
							subject_loyalty > 50
						}
						has_mutual_scripted_relation = {
							type = relation_type:alliance
							target = international_organization:hre.leader_country
						}
					}
				}
			}
			value = international_organization:hre.leader_country.country_color
		}
	}
	
	legend_key = {
		limit = {
			has_owner = yes
			OR = {
				owner = c:BOH
				owner = international_organization:hre.leader_country
			}
		}

		desc = "LOCATION_OWNER_NAME"
		color = owner.country_color
	}
}
