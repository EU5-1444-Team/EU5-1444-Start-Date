cb_throw_off_overlord = {
    create_visible = {
        any_international_organization = {
            international_organization_type = international_organization_type:european_tribal_confederation
            root = { is_member_of_international_organization = prev }
            any_country_with_special_status_of_type = {
                type = confederation_overlord
                this = scope:target
            }
        }
    }
    create_enabled = {
        any_international_organization = {
            international_organization_type = international_organization_type:european_tribal_confederation
            root = { is_member_of_international_organization = prev }
            any_country_with_special_status_of_type = {
                type = confederation_overlord
                this = scope:target
            }
        }
        NOT = { has_truce_with = scope:target }
    }
    war_goal_type = throw_off_overlord_wargoal
    ai_will_do = {
        add = {
            desc = "CB_REASON_IMPORTANT_CB"
            value = 100
        }
        # Less likely if obligations are light
        if = {
            limit = {
                any_international_organization = {
                    international_organization_type = international_organization_type:european_tribal_confederation
                    root = { is_member_of_international_organization = prev }
                    international_organization_has_policy = policy:european_tribal_low_tribute_policy
                }
            }
            subtract = {
                desc = "LOW_TRIBUTE_DISINCENTIVE"
                value = 75
            }
        }
        if = {
            limit = {
                any_international_organization = {
                    international_organization_type = international_organization_type:european_tribal_confederation
                    root = { is_member_of_international_organization = prev }
                    international_organization_has_policy = policy:european_tribal_high_tribute_policy
                }
            }
            add = {
                desc = "HIGH_TRIBUTE_INCENTIVE"
                value = 50
            }
        }
    }
}
cb_enforce_overlord = {
    create_visible = {
        any_international_organization = {
            international_organization_type = international_organization_type:european_tribal_confederation
            scope:target = { is_member_of_international_organization = prev }
            NOT = {
                any_international_organization_member = {
                    has_special_status_in_international_organization = {
                        type = special_status:confederation_overlord
                        international_organization = prev
                    }
                }
            }
        }
        OR = {
            is_neighbor_of = scope:target
            any_international_organization = {
                international_organization_type = international_organization_type:european_tribal_confederation
                any_international_organization_member = {
                    is_neighbor_of = root
                }
                NOT = {
                    any_international_organization_member = {
                        has_special_status_in_international_organization = {
                            type = special_status:confederation_overlord
                            international_organization = prev
                        }
                    }
                }
            }
        }
    }
    create_enabled = {
        NOT = { has_truce_with = scope:target }
    }
    war_goal_type = enforce_overlord_wargoal
    ai_will_do = {
        add = {
            desc = "CB_REASON_IMPORTANT_CB"
            value = 80
        }
    }
}
