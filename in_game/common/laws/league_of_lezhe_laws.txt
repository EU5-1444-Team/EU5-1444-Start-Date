league_of_lezhe_defense_law = {
	law_category = military
	type = international_organization
	potential = {
		international_organization_type = international_organization_type:league_of_lezhe
	}
	allow = {
		always = yes
	}
	locked = {
	}
	league_of_lezhe_defense_policy = {
		allow = {
			scope:actor ?= {
				NOT = {
					tag = VEN
				}
			}
		}
		# Albanian League members defend each other.
		join_defensive_wars_always = {
			custom_tooltip = {
				text = JOIN_WARS_DEFAULT_REASON
				always = yes
			}
			scope:actor ?= {
				NOT = {
					has_special_status_in_international_organization = {
						type = special_status:league_backer
						international_organization = root
					}
				}
			}
			international_organization:league_of_lezhe ?= {
				NOT = {
					any_country_with_special_status_of_type = {
						type = league_backer
					}
				}
			}
		}
		# If Venice remains in the League as Backer, it can be called only against the Ottomans.
		join_defensive_wars_auto_call = {
			custom_tooltip = {
				text = JOIN_WARS_DEFAULT_REASON
				always = yes
			}
			scope:actor ?= {
				NOT = {
					has_special_status_in_international_organization = {
						type = special_status:league_backer
						international_organization = root
					}
				}
			}
			international_organization:league_of_lezhe ?= {
				any_country_with_special_status_of_type = {
					type = league_backer
				}
			}
			scope:target ?= {
				this = c:TUR
			}
		}
		wants_propose_policy = {
			subtract = {
				value = 1000
			}
		}
	}
}

league_of_lezhe_venetian_governance_law = {
	law_category = administrative
	type = international_organization
	potential = {
		international_organization_type = international_organization_type:league_of_lezhe
	}
	allow = {
		always = yes
	}
	locked = {
	}
	no_venetian_local_governors_policy = {
		allow = {
			scope:actor ?= {
				NOT = {
					tag = VEN
				}
			}
		}
		wants_propose_policy = {
			subtract = {
				value = 1000
			}
		}
	}
	on_fully_activated = {
		if = {
			limit = {
				country_exists = c:VEN
				c:VEN = {
					is_member_of_international_organization = international_organization:league_of_lezhe
					any_owned_location = {
						area = area:albania_area
					}
				}
			}
			area:albania_area = {
				every_location_in_area = {
					limit = {
						owner = c:VEN
					}
					create_country_from_location = {
						make_subject_of = {
							target = c:VEN
							type = subject_type:governorate
						}
						change_country_name = lezhe_governorate_name
						change_country_adjective = lezhe_governorate_name_ADJ
						save_scope_as = lezhe_venetian_governor_country
					}
					if = {
						limit = {
							exists = scope:lezhe_venetian_governor_country
						}
						scope:lezhe_venetian_governor_country = {
							every_owned_location = {
								limit = {
									owner ?= scope:lezhe_venetian_governor_country
								}
								add_core = scope:lezhe_venetian_governor_country
								change_integration_level = core
							}
						}
					}
				}
			}
		}
	}
	months = 6
	venetian_local_governors_policy = {
		allow = {
			scope:actor ?= {
				NOT = {
					tag = VEN
				}
			}
			country_exists = c:VEN
			c:VEN = {
				is_member_of_international_organization = international_organization:league_of_lezhe
				opinion = {
					target = international_organization:league_of_lezhe.leader_country
					value >= 100
				}
				any_owned_location = {
					area = area:albania_area
				}
			}
		}
		wants_propose_policy = {
			add = {
				value = 120
			}
			if = {
				limit = {
					international_organization:league_of_lezhe = {
						international_organization_has_leader = yes
					}
					c:VEN = {
						opinion = {
							target = international_organization:league_of_lezhe.leader_country
							value >= 100
						}
					}
				}
				add = {
					value = 120
				}
			}
			else = {
				subtract = {
					value = 200
				}
			}
		}
		wants_this_policy_bias = {
			if = {
				limit = {
					scope:actor ?= { tag = VEN }
				}
				subtract = {
					desc = "POLICY_RESTRICTS_SOVEREIGNTY"
					value = 85
				}
				if = {
					limit = {
						international_organization:league_of_lezhe = {
							international_organization_has_leader = yes
						}
					}
					add = {
						desc = "POLICY_BIAS_OPINION_OF_IO_LEADER"
						value = "opinion(international_organization:league_of_lezhe.leader_country)"
					}
					add = {
						desc = "IO_LEADER_DIP_REP"
						value = international_organization:league_of_lezhe.leader_country.modifier:diplomatic_reputation
						multiply = 10
					}
				}
				if = {
					limit = {
						average_control_in_home_region < 0.55
					}
					add = {
						desc = "VOTE_BASE_VALUE"
						value = 35
					}
				}
				if = {
					limit = {
						monthly_balance < 0
					}
					add = {
						desc = "VOTE_BASE_VALUE"
						value = 25
					}
				}
				if = {
					limit = {
						any_owned_location = {
							area = area:albania_area
			province = { province_average_control < 0.6 }
					}
				}
				if = {
					limit = {
						any_owned_location = {
							area = area:albania_area
							province = { province_average_control > 0.85 }
						}
					}
					subtract = {
						desc = "VOTE_BASE_VALUE"
						value = 20
					}
				}
			}
			else = {
				add = {
					desc = "DIPLOREASON_BASE"
					value = 35
				}
				if = {
					limit = {
						international_organization:league_of_lezhe = {
							international_organization_has_leader = yes
						}
					}
					add = {
						desc = "POLICY_BIAS_OPINION_OF_IO_LEADER"
						value = "opinion(international_organization:league_of_lezhe.leader_country)"
						multiply = 0.25
					}
				}
			}
			add = {
				desc = "BIAS_FROM_RULER_TRAITS"
				value = modifier:bias_for_diplomat_policies
			}
		}
		on_fully_activated = {
			if = {
				limit = {
					country_exists = c:VEN
					c:VEN = {
						is_member_of_international_organization = international_organization:league_of_lezhe
						any_owned_location = {
							area = area:albania_area
						}
					}
				}
				area:albania_area = {
					every_location_in_area = {
						limit = {
							owner = c:VEN
						}
						create_country_from_location = {
							make_subject_of = {
								target = c:VEN
								type = subject_type:governorate
							}
							change_country_name = lezhe_governorate_name
							change_country_adjective = lezhe_governorate_name_ADJ
							save_scope_as = lezhe_venetian_governor_country
						}
						if = {
							limit = {
								exists = scope:lezhe_venetian_governor_country
							}
							scope:lezhe_venetian_governor_country = {
								every_owned_location = {
									limit = {
										owner ?= scope:lezhe_venetian_governor_country
									}
									add_core = scope:lezhe_venetian_governor_country
									change_integration_level = core
								}
							}
						}
					}
				}
			}
		}
		months = 6
	}
}

league_of_lezhe_unification_law = {
	law_category = centralization
	type = international_organization
	potential = {
		international_organization_type = international_organization_type:league_of_lezhe
	}
	allow = {
		any_international_organization_member = {
			count >= 2
			NOT = {
				has_special_status_in_international_organization = {
					type = special_status:league_backer
					international_organization = root
				}
			}
		}
	}
	custom_tags = {
		forbids_no_policy
	}
	preserve_league_of_lezhe_policy = {
		allow = {
			scope:actor ?= {
				NOT = {
					tag = VEN
				}
			}
		}
		wants_propose_policy = {
			subtract = {
				value = 1000
			}
		}
	}
	on_fully_activated = {
		custom_tooltip = league_of_lezhe_unification_tt
		every_international_organization_member = {
			limit = {
				NOT = {
					this = root.leader_country
				}
				NOT = {
					has_special_status_in_international_organization = {
						type = special_status:league_backer
						international_organization = root
					}
				}
			}
			root.leader_country = {
				annex_country = {
					country = prev
					reason = Diplomatic
				}
			}
		}
		root.leader_country = {
			form_country = formable_country:ALB_f
			if = {
				limit = {
					country_rank_level < 3
				}
				set_country_rank_effect = {
					rank = country_rank:rank_kingdom
				}
			}
		}
		leader_country = {
			destroy_international_organization = {
				target = international_organization:league_of_lezhe
			}
		}
	}
	years = 10
	unify_albania_policy = {
		allow = {
			scope:actor ?= {
				NOT = {
					tag = VEN
				}
			}
		}
		wants_propose_policy = {
			add = {
				value = 500
			}
		}
		wants_this_policy_bias = {
			add = {
				value = prev_opinion_of_this
				multiply = 0.3
			}
		}
		on_fully_activated = {
			custom_tooltip = league_of_lezhe_unification_tt
			every_international_organization_member = {
				limit = {
					NOT = {
						this = root.leader_country
					}
					NOT = {
						has_special_status_in_international_organization = {
							type = special_status:league_backer
							international_organization = root
						}
					}
				}
				root.leader_country = {
					annex_country = {
						country = prev
						reason = Diplomatic
					}
				}
			}
			root.leader_country = {
				form_country = formable_country:ALB_f
				if = {
					limit = {
						country_rank_level < 3
					}
					set_country_rank_effect = {
						rank = country_rank:rank_kingdom
					}
				}
			}
			leader_country = {
				destroy_international_organization = {
					target = international_organization:league_of_lezhe
				}
			}
		}
		years = 10
	}
}
