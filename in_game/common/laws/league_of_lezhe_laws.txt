league_of_lezhe_unification_law = {
	law_category = centralization
	type = international_organization
	potential = {
		international_organization_type = international_organization_type:league_of_lezhe
	}
	allow = {
		international_organization = {
			total_members >= 2
		}
	}
	custom_tags = {
		forbids_no_policy
	}
	preserve_league_of_lezhe_policy = {
		wants_propose_policy = {
			subtract = {
				value = 1000
			}
		}
	}
	unify_albania_policy = {
		wants_propose_policy = {
			add = {
				value = 500
			}
		}
		wants_this_policy_bias = {
			add = {
				value = prev_opinion_of_this
				multiply = 0.3
			}
			add = {
				value = root.relative_strength(scope:recipient)
				multiply = 50
			}
		}
		on_fully_activated = {
			custom_tooltip = league_of_lezhe_unification_tt
			root.leader_country = {
				form_country = formable_country:ALB_f
				if = {
					limit = {
						country_rank_level < 3
					}
					set_country_rank_effect = {
						rank = country_rank:rank_kingdom
					}
				}
			}
			every_international_organization_member = {
				limit = {
					NOT = {
						this = root.leader_country
					}
				}
				root.leader_country = {
					annex_country = {
						country = prev
						reason = Diplomatic
					}
				}
			}
			disband_international_organization = yes
		}
		years = 25
	}
}
