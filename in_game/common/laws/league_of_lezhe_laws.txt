league_of_lezhe_defense_law = {
	law_category = military
	type = international_organization
	potential = {
		international_organization_type = international_organization_type:league_of_lezhe
	}
	allow = {
		always = yes
	}
	locked = {
	}
	league_of_lezhe_defense_policy = {
		allow = {
			NOT = {
				tag = VEN
			}
		}
		# Albanian League members defend each other.
		join_defensive_wars_always = {
			scope:actor ?= {
				NOT = {
					has_special_status_in_international_organization = {
						type = special_status:league_backer
						international_organization = root
					}
				}
			}
			scope:recipient ?= {
				NOT = {
					has_special_status_in_international_organization = {
						type = special_status:league_backer
						international_organization = root
					}
				}
			}
		}
		# If Venice remains in the League as Backer, it can be called only against the Ottomans.
		join_defensive_wars_auto_call = {
			scope:actor ?= {
				NOT = {
					has_special_status_in_international_organization = {
						type = special_status:league_backer
						international_organization = root
					}
				}
			}
			scope:recipient ?= {
				has_special_status_in_international_organization = {
					type = special_status:league_backer
					international_organization = root
				}
			}
			scope:target ?= {
				this = c:TUR
			}
		}
		wants_propose_policy = {
			subtract = {
				value = 1000
			}
		}
	}
}

league_of_lezhe_unification_law = {
	law_category = centralization
	type = international_organization
	potential = {
		international_organization_type = international_organization_type:league_of_lezhe
	}
	allow = {
		any_international_organization_member = {
			count >= 2
			NOT = {
				has_special_status_in_international_organization = {
					type = special_status:league_backer
					international_organization = root
				}
			}
		}
	}
	custom_tags = {
		forbids_no_policy
	}
	preserve_league_of_lezhe_policy = {
		allow = {
			NOT = {
				tag = VEN
			}
		}
		wants_propose_policy = {
			subtract = {
				value = 1000
			}
		}
	}
	unify_albania_policy = {
		allow = {
			NOT = {
				tag = VEN
			}
		}
		wants_propose_policy = {
			add = {
				value = 500
			}
		}
		wants_this_policy_bias = {
			add = {
				value = prev_opinion_of_this
				multiply = 0.3
			}
		}
		on_fully_activated = {
			custom_tooltip = league_of_lezhe_unification_tt
			root.leader_country = {
				form_country = formable_country:ALB_f
				if = {
					limit = {
						country_rank_level < 3
					}
					set_country_rank_effect = {
						rank = country_rank:rank_kingdom
					}
				}
			}
			every_international_organization_member = {
				limit = {
					NOT = {
						this = root.leader_country
					}
					NOT = {
						has_special_status_in_international_organization = {
							type = special_status:league_backer
							international_organization = root
						}
					}
				}
				root.leader_country = {
					annex_country = {
						country = prev
						reason = Diplomatic
					}
				}
			}
			destroy_international_organization = {
				target = root
			}
		}
		years = 25
	}
}
