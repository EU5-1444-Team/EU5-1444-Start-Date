	law_category = centralization
	type = international_organization

		international_organization_type = international_organization_type:frisian_freedom
	}
	custom_tags = {
		forbids_no_policy
	}
	preserve_frisian_freedom_policy = {
		allow = {
			always = yes
		}
		wants_propose_policy = {
			subtract = {
				value = 1000
			}
		}
	}
	unify_frisia_policy = {
		allow = {
			always = yes
		}
		wants_propose_policy = {
			add = {
				value = 500
			}
		}
		wants_this_policy_bias = {
			add = {
				value = prev_opinion_of_this
				multiply = 0.3
			}
		}
		on_fully_activated = {
			custom_tooltip = frisian_freedom_unification_tt
			leader_country = {
				trigger_event_non_silently = { id = io_unification.100 }
			}
		}
		years = 10
	}
}
