frisian_freedom_unification_law = {
	law_category = centralization
	type = international_organization
	potential = {
		international_organization_type = international_organization_type:frisian_freedom
	}
	allow = {
		any_international_organization_member = {
			count >= 2
		}
	}
	custom_tags = {
		forbids_no_policy
	}
	preserve_frisian_freedom_policy = {
		allow = {
			always = yes
		}
		wants_propose_policy = {
			subtract = {
				value = 1000
			}
		}
	}
	unify_frisia_policy = {
		allow = {
			always = yes
		}
		wants_propose_policy = {
			add = {
				value = 500
			}
		}
		wants_this_policy_bias = {
			add = {
				value = prev_opinion_of_this
				multiply = 0.3
			}
		}
		on_fully_activated = {
			custom_tooltip = frisian_freedom_unification_tt
			every_international_organization_member = {
				limit = {
					NOT = {
						this = root.leader_country
					}
				}
				root.leader_country = {
					annex_country = {
						country = prev
						reason = Diplomatic
					}
				}
			}
			root.leader_country = {
				form_country = formable_country:FRI_f
				if = {
					limit = {
						country_rank_level < 3
					}
					set_country_rank_effect = {
						rank = country_rank:rank_kingdom
					}
				}
			}
			leader_country = {
				destroy_international_organization = {
					target = international_organization:frisian_freedom
				}
			}
		}
		years = 10
	}
}
