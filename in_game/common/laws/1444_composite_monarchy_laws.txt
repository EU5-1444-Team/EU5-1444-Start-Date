#Composite Monarchy Foundation Law
composite_monarchy_foundation_law = {
	type = international_organization
	law_category = foundation
	potential = { 
		international_organization_type = international_organization_type:composite_monarchy
	}
	allow = {
		international_organization_lifetime >= 5
	}
	locked = {
		international_organization_has_law = law:composite_monarchy_foundation_law
		international_organization_has_policy = policy:composite_maintain_autonomy_policy
	}
	custom_tags = { forbids_no_policy }

	composite_maintain_autonomy_policy = {
		wants_propose_policy = {
			subtract = {
				desc = "VOTE_BASE_VALUE"
				value = 100
			}
		}
	}

	composite_unite_crowns_policy = {
		allow = {
			scope:recipient = { leader_country ?= { exists = root } }
		}
		on_activate = {
			change_composite_integration = { value = 5 }
		}
		on_deactivate = {
			change_composite_integration = { value = -5 }
		}
		country_modifier = {
			monthly_legitimacy = 0.05
		}
		wants_propose_policy = {
			if = {
				limit = {
					scope:recipient = { leader_country ?= root }
				}
				add = {
					desc = "VOTE_BASE_VALUE"
					value = 50
				}
			}
			else = {
				subtract = {
					desc = "VOTE_BASE_VALUE"
					value = 50
				}
			}
		}
		wants_this_policy_bias = {
			add = {
				desc = "VOTE_BASE_VALUE"
				value = -25
			}
		}
		wants_keep_policy = {
			add = {
				desc = "VOTE_CONSERVATIVE_CLIMATE"
				value = 50
			}
		}
	}
}

				}

			}

			}
		}
		wants_keep_policy = {
			add = {
				desc = "DIPLOREASON_BASE"
				value = 20
			}
