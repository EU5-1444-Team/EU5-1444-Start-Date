mongol_local_administration = {
    type = international_organization
    law_category = administrative
    requires_vote = yes
    potential = {
        OR = {
            international_organization_type = international_organization_type:khalkha_confederation
            international_organization_type = international_organization_type:khorchin_confederation
            international_organization_type = international_organization_type:nogai_confederation
            international_organization_type = international_organization_type:bashkir_confederation
        }
    }
    allow = {
        leader_country ?= {
            societal_value:centralization_vs_decentralization < 90
        }
    }
    locked = {
        custom_tooltip = {
            text = 5_years_cd
            leader_country ?= {
                has_variable = local_adm_law_cooldown_variable
            }
        }
    }
 
    mongol_tribal_administration_policy = {
        wants_propose_policy = { add = { value = 100 } }
        wants_this_policy_bias = {
            add = { desc = "POLICY_BIAS_TRIBAL_TRADITION" value = 25 }
        }
        on_activate = {
            leader_country ?= {
                set_variable = {
                    name = local_adm_law_cooldown_variable
                    value = yes
                    years = 5
                }
                add_tribal_cohesion = tribal_cohesion_weak_penalty
                if = {
                    limit = { ruler_or_regent ?= { has_variable = difficult_sinicization_variable } }
                    add_tribal_cohesion = tribal_cohesion_mild_penalty
                    add_stability = stability_mild_penalty
                }
            }
            every_international_organization_member = {
                change_societal_value = {
                    type = centralization_vs_decentralization
                    value = societal_value_minor_move_to_right
                }
            }
        }
        years = 5
        country_modifier = {
            monthly_horde_unity = 0.05
            discipline = 0.0075
            global_estate_satisfaction_recovery = 0.02
            stability_decay = 0.001
