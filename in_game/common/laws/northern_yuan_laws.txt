kurultai_law = {
    law_category = administrative
    type = international_organization
    potential = {
        international_organization_type = international_organization_type:northern_yuan

    allow = {
        international_organization:northern_yuan = {
            var:yeke_mongghul_ulus >= 80

    }
    has_levels = yes
    custom_tags = { forbids_no_policy }
 
    no_kurultai_policy = {
        level = 1
        wants_propose_policy = {
            add = { desc = "POLICY_BIAS_LEADER_WANTS" value = 0 }
            if = {
                limit = {
                    has_special_status_in_international_organization = {
                        type = special_status:protector_general
                        international_organization = international_organization:northern_yuan

        wants_this_policy_bias = {
            add = { desc = "POLICY_ANCIENT_TRADITION" value = 25 }

                limit = {
