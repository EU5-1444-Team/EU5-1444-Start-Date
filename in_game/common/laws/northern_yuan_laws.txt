kurultai_law = {
    law_category = administrative
    type = international_organization
    requires_vote = yes
    potential = {
        international_organization_type = international_organization_type:northern_yuan
    }
    allow = {
        international_organization:northern_yuan = {
            var:yeke_mongghul_ulus >= 80
        }
    }
    has_levels = yes
    custom_tags = { forbids_no_policy }

    no_kurultai_policy = {
        on_activate = {
            international_organization:northern_yuan = {
                change_variable = { name = yeke_mongghul_ulus add = -80 }
            }
        }
        wants_propose_policy = {
            subtract = { desc = "POLICY_BIAS_LEADER_OPPOSES" value = 500 }
        }
        wants_this_policy_bias = {
            add = { desc = "POLICY_ANCIENT_TRADITION" value = 50 }
        }
    }

    formal_kurultai_policy = {
        on_activate = {
            international_organization:northern_yuan = {
                change_variable = { name = yeke_mongghul_ulus add = -80 }
            }
        }

        # Taishi
        country_modifier = {
            potential_trigger = {
                has_special_status_in_international_organization = {
                    type = special_status:protector_general
                    international_organization = international_organization:northern_yuan
                }
            }
            monthly_prestige = 0.15
            diplomatic_reputation = diplomatic_reputation_mild_bonus
            diplomatic_capacity = 2
            max_diplomats = 1
        }

        # Yuan Emperor
        country_modifier = {
            potential_trigger = {
                is_leader_of_international_organization = international_organization:northern_yuan
                NOT = {
                    has_special_status_in_international_organization = {
                        type = special_status:protector_general
                        international_organization = international_organization:northern_yuan
                    }
                }
            }
            monthly_prestige = 0.05
            diplomatic_reputation = diplomatic_reputation_weak_bonus
            monthly_legitimacy = -0.05
            monthly_devotion = -0.05
            monthly_horde_unity = -0.05
            monthly_tribal_cohesion = -0.05
            monthly_republican_tradition = -0.05
        }

        # Members
        country_modifier = {
            potential_trigger = {
                NOT = { is_leader_of_international_organization = international_organization:northern_yuan }
                NOT = {
                    has_special_status_in_international_organization = {
                        type = special_status:protector_general
                        international_organization = international_organization:northern_yuan
                    }
                }
            }
            monthly_prestige = 0.03
            diplomatic_reputation = diplomatic_reputation_weak_bonus
        }

        wants_propose_policy = {
            add = { desc = "POLICY_BIAS_LEADER_WANTS" value = 50 }
            if = {
                limit = {
                    has_special_status_in_international_organization = {
                        type = special_status:protector_general
                        international_organization = international_organization:northern_yuan
                    }
                }
                if = {
                    limit = {
                        ruler_or_heir_if_regent ?= {
                            OR = {
                                dynasty ?= dynasty:borjigin_dynasty
                                dynasty ?= dynasty:jochi_dynasty
                                dynasty ?= dynasty:giray_dynasty
                                dynasty ?= dynasty:chagatayid_dynasty
                                dynasty ?= dynasty:shaybanid_dynasty
                            }
                        }
                    }
                    add = { desc = "POLICY_GENGHISID_PREFERS_YUAN" value = 25 }
                }
                else = {
                    subtract = { desc = "POLICY_NON_GENGHISID_OPPOSES_YUAN" value = 25 }
                }
            }
        }
        wants_this_policy_bias = {
            add = { desc = "POLICY_MILD_CENTRALIZATION" value = 5 }
        }
    }
