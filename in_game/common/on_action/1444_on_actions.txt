on_game_start = {
	on_actions = {
		1444_staging_events
	}
}

# root = winner, scope:location = location, scope:winner = new owner, scope:loser = old owner
on_took_location_in_peace_treaty = {
	on_actions = {
		1444_sengoku_border_cb_from_peace
	}
}

1444_sengoku_border_cb_from_peace = {
	trigger = {
		is_situation_active = situation:sengoku
		exists = scope:winner
	}
	effect = {
		scope:winner = {
			1444_refresh_sengoku_neighbor_cbs = yes
		}
	}
}

# root = annexing country, scope:target = annexed country
on_annexed = {
	on_actions = {
		1444_sengoku_border_cb_from_annex
	}
}

1444_sengoku_border_cb_from_annex = {
	trigger = {
		is_situation_active = situation:sengoku
	}
	effect = {
		1444_refresh_sengoku_neighbor_cbs = yes
	}
}

# root = country, scope:actor = country who declares the war, scope:recipient = country who gets declared on, scope:war = war
on_war_declared = {
	effect = {
		if = {
			limit = {
				scope:war ?= {
					casus_belli ?= casus_belli:cb_shan_confederation_raid
				}
			}
			trigger_event_non_silently = 1444_shan_confederation.1
		}
	}
}

1444_staging_events = {
	events = {
		delay = {
			days = 1
		}
		1444_flavor_tur.1
	}
}
