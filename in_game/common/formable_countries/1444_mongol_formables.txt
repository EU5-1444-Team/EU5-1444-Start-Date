	level = 3
	required_locations_fraction = 0.3
	rule = historical
	
	potential = {
		or = {
			always = yes
			or = {
				tag = AGZ
				tag = KGZ
				tag = MGZ
			}
		}
	}

	allow = {
		owns = location:sarai_al_jadid
		or = {
			AND = {
				tag = AGZ
				NOT = { country_exists = c:KGZ }
				NOT = { country_exists = c:MGZ }
			}
			AND = {
				tag = KGZ
				NOT = { country_exists = c:AGZ }
				NOT = { country_exists = c:MGZ }
			}
			AND = {
				tag = MGZ
				NOT = { country_exists = c:AGZ }
				NOT = { country_exists = c:KGZ }
			}
		}
	}
	
	name = GTH

	color = map_GTH
	regions = {
		steppes_region
	}
	form_effect = {
		set_country_rank_effect = { rank = country_rank:rank_kingdom }
		set_capital = location:sarai_al_jadid
	}
}

			}
			AND = {

				NOT = { country_exists = c:SBR }
			}
			AND = {
				tag = SBR
